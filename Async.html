// ========================== // setTimeout() Notes // ========================== // setTimeout() executes the callback ONLY ONCE // after the given delay (milliseconds). const bg = setTimeout(() => { document.body.style.background = "red"; console.log("hi"); }, 3000); // bg stores the TIMER ID, NOT the function. // clearTimeout() needs this timer ID to cancel the timer. clearTimeout(bg); // Since we cancelled immediately, // background will NEVER become red. // "hi" will NEVER print. // ========================== // Function Reference // ========================== const saySohail = () => { // console.log("Sohail"); } // IMPORTANT: // We pass the FUNCTION REFERENCE. // DON'T write saySohail() setTimeout(saySohail, 1000); // Wrong ❌ // setTimeout(saySohail(),1000) // ========================== // Button Example // ========================== const btn = document.getElementById("start"); btn.addEventListener("click", () => { // Timer starts ONLY after clicking the button. setTimeout(() => { console.log("Welcome Sohail"); }, 3000); }); // ========================== // Change Text Example // ========================== const changeText = function () { document.querySelector("h1").textContent = "hello sohail"; }; // Store timer ID. const textTime = setTimeout(changeText, 3000); // ========================== // Stop Button // ========================== document.getElementById("stop").addEventListener("click", function () { // clearTimeout() cancels ONLY // if the timer has NOT executed yet. clearTimeout(textTime); console.log("STOPPED!"); }); // ====================================== // GOLDEN RULES (Must Remember) // ====================================== // 1. // setTimeout() runs ONLY ONE TIME. // 2. // Delay is always in milliseconds. // 1000 = 1 second. // 3. // Pass FUNCTION REFERENCE. // ✔ setTimeout(show,2000) // ✖ setTimeout(show(),2000) // 4. // setTimeout() returns a TIMER ID. // 5. // clearTimeout() accepts ONLY TIMER ID. // ✔ const id = setTimeout(show,2000); clearTimeout(id); // ✖ clearTimeout(show); // 6. // If timer already executed, // clearTimeout() does NOTHING. // 7. // Timeline // Create Timer // ↓ // Wait // ↓ // Execute Callback // // OR // // Create Timer // ↓ // clearTimeout() // ↓ // Callback NEVER runs.
